1. プロダクト概要

1.1 コンセプト

日記を書くと AI が感情を分析し、感情の推移や傾向を可視化するジャーナルアプリ。 「書く → 分析される → 自分の変化が見える」のサイクルでユーザーの自己理解と習慣化を支援する。

1.2 ターゲットユーザー

  • 自己分析・内省に関心のある20〜40代
  • 日記習慣を作りたいが続かない人
  • メンタルヘルスに関心があり、感情の変化を客観視したい人

4.2 テーブル定義

profiles

CREATE TABLE profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  display_name TEXT,
  avatar_url TEXT,
  timezone TEXT DEFAULT 'Asia/Tokyo',
  free_entries_used INT DEFAULT 0,
  free_analyses_used INT DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

備考: - auth.users の作成時にトリガーで自動生成 - free_entries_used / free_analyses_used は月初にcronジョブでリセット

journal_entries

CREATE TABLE journal_entries (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES profiles(id) ON DELETE CASCADE,
  content TEXT NOT NULL,
  tags TEXT[] DEFAULT '{}',
  template_type TEXT DEFAULT 'free',
  entry_date DATE NOT NULL DEFAULT CURRENT_DATE,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_entries_user_id ON journal_entries(user_id);
CREATE INDEX idx_entries_entry_date ON journal_entries(user_id, entry_date);
CREATE INDEX idx_entries_tags ON journal_entries USING GIN(tags);

備考: - entry_date と created_at は分離（深夜に「今日の分」を書くケース対応） - tags は PostgreSQL の text[] 型、GINインデックスで高速検索 - template_type: 'free', 'gratitude', 'goal', 'reflection', 'custom'

emotion_analyses

CREATE TABLE emotion_analyses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  entry_id UUID NOT NULL UNIQUE REFERENCES journal_entries(id) ON DELETE CASCADE,
  joy REAL DEFAULT 0 CHECK (joy >= 0 AND joy <= 1),
  anger REAL DEFAULT 0 CHECK (anger >= 0 AND anger <= 1),
  sadness REAL DEFAULT 0 CHECK (sadness >= 0 AND sadness <= 1),
  surprise REAL DEFAULT 0 CHECK (surprise >= 0 AND surprise <= 1),
  calm REAL DEFAULT 0 CHECK (calm >= 0 AND calm <= 1),
  energy_level INT DEFAULT 3 CHECK (energy_level >= 1 AND energy_level <= 5),
  ai_feedback TEXT,
  raw_response JSONB,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_analyses_entry_id ON emotion_analyses(entry_id);

備考: - 各感情スコアは 0.0〜1.0 の float（合計は1.0に正規化しない。複数感情が同時に高いケースを許容） - energy_level は 1（低）〜5（高）の5段階 - raw_response に Claude API のレスポンス全体を JSONB で保存（デバッグ・改善用） - 1エントリにつき1分析（entry_id に UNIQUE制約）

streaks

CREATE TABLE streaks (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL UNIQUE REFERENCES profiles(id) ON DELETE CASCADE,
  current_streak INT DEFAULT 0,
  longest_streak INT DEFAULT 0,
  last_entry_date DATE,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

備考: - ユーザーごとに1レコード（UNIQUE制約） - 日記投稿時に last_entry_date との差分で current_streak を更新 - 差分 = 1日 → current_streak += 1 - 差分 = 0日（同日2回目）→ 変更なし - 差分 > 1日 → current_streak = 1（リセット） - longest_streak = MAX(current_streak, longest_streak)

subscriptions

CREATE TABLE subscriptions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL UNIQUE REFERENCES profiles(id) ON DELETE CASCADE,
  stripe_customer_id TEXT UNIQUE,
  stripe_subscription_id TEXT UNIQUE,
  status TEXT DEFAULT 'free' CHECK (status IN ('free', 'active', 'canceled', 'past_due', 'incomplete')),
  plan TEXT DEFAULT 'free' CHECK (plan IN ('free', 'pro')),
  current_period_start TIMESTAMPTZ,
  current_period_end TIMESTAMPTZ,
  cancel_at_period_end BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

備考: - Stripe Webhook で status を同期 - cancel_at_period_end = true の場合、期間終了まではPro機能を維持 - plan はアプリ内での権限判定に使用

monthly_reports

CREATE TABLE monthly_reports (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES profiles(id) ON DELETE CASCADE,
  year_month TEXT NOT NULL,
  emotion_summary JSONB,
  ai_narrative TEXT,
  total_entries INT DEFAULT 0,
  avg_energy REAL,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(user_id, year_month)
);

CREATE INDEX idx_reports_user_month ON monthly_reports(user_id, year_month);

備考: - year_month は '2026-11' 形式 - emotion_summary 例: json { "avg_joy": 0.65, "avg_anger": 0.12, "avg_sadness": 0.20, "avg_surprise": 0.30, "avg_calm": 0.55, "dominant_emotion": "joy", "trend": "improving" } - 月初にcronで前月分を自動生成（Proユーザーのみ）

4.3 RLS（Row Level Security）ポリシー

全テーブルに対して以下のポリシーを適用:

-- 全テーブル共通パターン
ALTER TABLE <table_name> ENABLE ROW LEVEL SECURITY;

-- SELECT: 自分のデータのみ閲覧可能
CREATE POLICY "Users can view own data"
  ON <table_name> FOR SELECT
  USING (user_id = auth.uid());

-- INSERT: 自分のデータのみ作成可能
CREATE POLICY "Users can insert own data"
  ON <table_name> FOR INSERT
  WITH CHECK (user_id = auth.uid());

-- UPDATE: 自分のデータのみ更新可能
CREATE POLICY "Users can update own data"
  ON <table_name> FOR UPDATE
  USING (user_id = auth.uid());

-- DELETE: 自分のデータのみ削除可能
CREATE POLICY "Users can delete own data"
  ON <table_name> FOR DELETE
  USING (user_id = auth.uid());

profiles テーブルは id = auth.uid() で判定（user_id ではなく id）。

4.4 Database Functions / Triggers

-- 1. 新規ユーザー登録時にprofilesを自動作成
CREATE OR REPLACE FUNCTION handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO profiles (id, display_name)
  VALUES (NEW.id, COALESCE(NEW.raw_user_meta_data->>'name', NEW.email));

  INSERT INTO streaks (user_id)
  VALUES (NEW.id);

  INSERT INTO subscriptions (user_id)
  VALUES (NEW.id);

  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION handle_new_user();

-- 2. updated_atの自動更新
CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER set_updated_at
  BEFORE UPDATE ON profiles
  FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER set_updated_at
  BEFORE UPDATE ON journal_entries
  FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER set_updated_at
  BEFORE UPDATE ON subscriptions
  FOR EACH ROW EXECUTE FUNCTION update_updated_at();

7.3 Free制限ロジック

// 日記作成時の制限チェック
async function canCreateEntry(userId: string): Promise<boolean> {
  const sub = await getSubscription(userId);
  if (sub.plan === 'pro') return true;

  const profile = await getProfile(userId);
  return profile.free_entries_used < 10;
}

// AI分析の制限チェック
async function canAnalyze(userId: string): Promise<boolean> {
  const sub = await getSubscription(userId);
  if (sub.plan === 'pro') return true;

  const profile = await getProfile(userId);
  return profile.free_analyses_used < 5;
}

// 月初リセット（Supabase Edge Function / cron）
// 毎月1日 00:00 JST に実行
async function resetMonthlyLimits() {
  await supabase
    .from('profiles')
    .update({ free_entries_used: 0, free_analyses_used: 0 })
    .neq('id', null); // 全ユーザー
}
